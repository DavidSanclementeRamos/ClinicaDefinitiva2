/**
 * Emite la factura: transiciona de BORRADOR → PENDIENTE.
 *
 * Reglas aplicadas:
 * - RN-INVOICE-001: Debe tener al menos un ítem
 * - RN-INVOICE-002: Total mayor a cero
 * - RN-INVOICE-007: Contrato requerido para aseguradoras/EPS
 * - RN-INVOICE-010: Factura emitida no puede modificarse
 *
 * Uso del generador:
 * - Se obtiene un número de factura válido desde InvoiceNumberGenerator.
 * - Se asigna al agregado y se registra el evento de emisión.
 */
public void emit(InvoiceNumberGenerator generator) {
    Objects.requireNonNull(generator, "InvoiceNumberGenerator is required");

    validateBeforeEmit();

    String previousState = this.status.getName();
    this.number = generator.next(); // Genera y valida el número de factura
    this.status = this.status.transitionTo(InvoiceStatus.Status.PENDING);
    this.updatedAt = LocalDateTime.now();

    // Evento de dominio enriquecido
    events.add(new EventosFacturacion.InvoiceStateChanged(
            this.id.toString(),
            previousState,
            this.status.getName(),
            this.number.getValue(),
            this.updatedAt
    ));
}


// Application Service
public class InvoiceApplicationService {

    private final InvoiceRepository repository;
    private final InvoiceNumberGenerator generator;

    public InvoiceApplicationService(InvoiceRepository repository) {
        this.repository = repository;
        this.generator = new InvoiceNumberGenerator("FAC", 0); // Prefijo FAC, secuencia inicial 0
    }

    public void emitInvoice(Invoice invoice) {
        invoice.emit(generator);
        repository.save(invoice);
        invoice.getEvents().forEach(eventPublisher::publish);
    }
}


public final class Rate {
    private final RateId id;
    private final LocalDateTime validFrom;
    private final LocalDateTime validUntil;

    public boolean isValidAt(LocalDateTime date) {
        return (date.isEqual(validFrom) || date.isAfter(validFrom))
               && (validUntil == null || date.isBefore(validUntil));
    }
}


public final class InvoiceDomainService {

    private final RateRepository rateRepository;

    public InvoiceDomainService(RateRepository rateRepository) {
        this.rateRepository = rateRepository;
    }

    public void validateNoExpiredRates(Invoice invoice) {
        for (InvoiceItem item : invoice.getItems()) {
            Rate rate = rateRepository.findById(item.getRateId())
                .orElseThrow(() -> new BusinessRuleViolationException(
                    BillingError.ERR_RATE_NOT_FOUND,
                    EntityContext.INVOICE,
                    "Rate not found for item: " + item.getId()
                ));

            if (!rate.isValidAt(invoice.getIssuedAt())) {
                throw new BusinessRuleViolationException(
                    BillingError.ERR_INVOICE_EXPIRED_RATE,
                    EntityContext.INVOICE,
                    "Invoice contains expired rate for item: " + item.getId()
                );
            }
        }
    }
}
